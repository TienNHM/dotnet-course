#### Bài 2 — Controller, validation và xử lý lỗi (Trung bình, 35 điểm)

`CustomersController` ở `api/v1/customers` với:

- `GET /` nhận `[FromQuery] GetCustomersRequest` (`Page` từ 1, `PageSize` từ 1 tới 100, `Search` tuỳ chọn), trả `PagedResponse<Customer>`.
- `GET /{id:int}` trả 200, hoặc **404 bằng giá trị trả về** `NotFound()` khi không có — không ném exception cho trường hợp thường gặp này ([bài 8.8](8.8-global-exception-handling.md)).
- `POST /` nhận `CreateCustomerRequest` (tên bắt buộc 2–200 ký tự, email hợp lệ, thông điệp lỗi tiếng Việt), trả 201 kèm `Location` qua `CreatedAtRoute`.
- `GlobalExceptionHandler` trả `ProblemDetails` 500 chỉ có `title`, `status`, `instance`, `traceId`. Để thử, map `GET /api/diagnostics/error` ném `InvalidOperationException` (chỉ ở Development).

**Tiêu chí chấm (kiểm tra được):**

- `GET /api/v1/customers/999` → 404, `Content-Type: application/problem+json`.
- `GET /api/v1/customers?pageSize=500` → 400 với lỗi ở khoá `PageSize`.
- `POST` với email sai → 400, body chứa `Email không hợp lệ`.
- `POST` hợp lệ → 201, có header `Location`.
- `GET /api/diagnostics/error` → 500, body không chứa thông điệp exception hay stack trace, có `traceId`.
- Không có `if (!ModelState.IsValid)` và không có `try/catch` nào trong controller.

```csharp
// Models/CustomerStore.cs — dữ liệu trong bộ nhớ, đăng ký Singleton
namespace CrmApi.Models;

public class CustomerStore
{
    private readonly Lock _lock = new();       // Singleton dùng chung cho mọi request -> phải khoá
    private readonly List<Customer> _customers =
    [
        new() { Id = 1, Name = "Nguyễn Văn An", Email = "an@example.com", Phone = "0901234567", CreatedAt = DateTime.UtcNow },
        new() { Id = 2, Name = "Trần Thị Bình", Email = "binh@example.com", CreatedAt = DateTime.UtcNow },
    ];

    public PagedResponse<Customer> GetPage(GetCustomersRequest request)
    {
        lock (_lock)
        {
            var query = _customers.AsEnumerable();
            if (!string.IsNullOrWhiteSpace(request.Search))
                query = query.Where(c =>
                    c.Name.Contains(request.Search, StringComparison.OrdinalIgnoreCase) ||
                    c.Email.Contains(request.Search, StringComparison.OrdinalIgnoreCase));

            var matched = query.ToList();
            var items = matched.Skip((request.Page - 1) * request.PageSize).Take(request.PageSize).ToList();
            return new PagedResponse<Customer>(items, request.Page, request.PageSize, matched.Count);
        }
    }

    public Customer? Find(int id)
    {
        lock (_lock) return _customers.FirstOrDefault(c => c.Id == id);
    }

    public Customer Add(CreateCustomerRequest request)
    {
        lock (_lock)
        {
            var customer = new Customer
            {
                Id = _customers.Max(c => c.Id) + 1,
                Name = request.Name!.Trim(),
                Email = request.Email!,
                Phone = request.Phone,
                CreatedAt = DateTime.UtcNow
            };
            _customers.Add(customer);
            return customer;
        }
    }
}

// Middleware/Middleware.cs
using System.Diagnostics;
using Microsoft.AspNetCore.Diagnostics;
using Microsoft.AspNetCore.Mvc;

namespace CrmApi.Middleware;

public class RequestTimingMiddleware(RequestDelegate next, ILogger<RequestTimingMiddleware> logger)
{
    public async Task InvokeAsync(HttpContext context)
    {
        var start = Stopwatch.GetTimestamp();
        try
        {
            await next(context);
        }
        finally
        {
            logger.LogInformation("HTTP {Method} {Path} -> {StatusCode} trong {Elapsed:0.0}ms",
                context.Request.Method, context.Request.Path, context.Response.StatusCode,
                Stopwatch.GetElapsedTime(start).TotalMilliseconds);
        }
    }
}

public class CorrelationIdMiddleware(RequestDelegate next, ILogger<CorrelationIdMiddleware> logger)
{
    private const string Header = "X-Correlation-ID";

    public async Task InvokeAsync(HttpContext context)
    {
        var correlationId = context.Request.Headers[Header].FirstOrDefault()
                            ?? Guid.NewGuid().ToString();

        context.Response.OnStarting(() =>
        {
            context.Response.Headers[Header] = correlationId;
            return Task.CompletedTask;
        });

        using (logger.BeginScope(new Dictionary<string, object> { ["CorrelationId"] = correlationId }))
        {
            await next(context);
        }
    }
}

public sealed class GlobalExceptionHandler(
    ILogger<GlobalExceptionHandler> logger,
    IProblemDetailsService problemDetailsService) : IExceptionHandler
{
    public async ValueTask<bool> TryHandleAsync(HttpContext context, Exception exception, CancellationToken ct)
    {
        logger.LogError(exception, "Lỗi chưa xử lý tại {Path}", context.Request.Path);

        context.Response.StatusCode = StatusCodes.Status500InternalServerError;
        return await problemDetailsService.TryWriteAsync(new ProblemDetailsContext
        {
            HttpContext = context,
            Exception = exception,
            ProblemDetails = new ProblemDetails
            {
                Status = StatusCodes.Status500InternalServerError,
                Title = "Lỗi hệ thống",
                Instance = context.Request.Path
            }
        });
    }
}

// Controllers/CustomersController.cs
using CrmApi.Models;
using Microsoft.AspNetCore.Mvc;

namespace CrmApi.Controllers;

[ApiController]
[Route("api/v1/customers")]
public class CustomersController(CustomerStore store) : ControllerBase
{
    [HttpGet]
    public ActionResult<PagedResponse<Customer>> GetAll([FromQuery] GetCustomersRequest request)
        => store.GetPage(request);

    [HttpGet("{id:int}", Name = "GetCustomerById")]
    [ProducesResponseType<Customer>(StatusCodes.Status200OK)]
    [ProducesResponseType(StatusCodes.Status404NotFound)]
    public ActionResult<Customer> GetById(int id)
    {
        var customer = store.Find(id);
        return customer is null ? NotFound() : customer;
    }

    [HttpPost]
    [ProducesResponseType<Customer>(StatusCodes.Status201Created)]
    public ActionResult<Customer> Create(CreateCustomerRequest request)
    {
        var customer = store.Add(request);
        return CreatedAtRoute("GetCustomerById", new { id = customer.Id }, customer);
    }
}

        Assert.Contains("traceId", body);
    }

    [Fact]
    public async Task CorrelationId_IsEchoed()
    {
        var request = new HttpRequestMessage(HttpMethod.Get, "/health/live");
        request.Headers.Add("X-Correlation-ID", "abc-123");

        var response = await factory.CreateClient().SendAsync(request);

        Assert.Equal("abc-123", response.Headers.GetValues("X-Correlation-ID").Single());
    }
}
```

`WebApplicationFactory` chạy ứng dụng ở môi trường Development, nên endpoint `/api/diagnostics/error` có mặt trong test.

</details>

## Tự kiểm tra

<details>
<summary>Vì sao lab yêu cầu trả 404 bằng NotFound() thay vì ném NotFoundException?</summary>

Vì "không tìm thấy" là trường hợp thường gặp, không phải bất thường. Ném exception tốn kém hơn nhiều, sinh log vô ích và đi đường vòng qua exception handler. Exception dành cho lỗi thật sự bất thường; handler toàn cục là lưới an toàn cho chúng.

</details>

<details>
<summary>Vì sao UseExceptionHandler phải đứng trước hai middleware tự viết?</summary>

Vì nó chỉ bắt được exception từ các lớp bên trong nó. Đặt sau thì lỗi trong middleware đo thời gian hay correlation ID không được xử lý và client nhận phản hồi 500 trống.

</details>

<details>
<summary>Test xanh có chứng minh API đúng không?</summary>

Không. Test chỉ có giá trị khi bạn đã thấy nó đỏ lúc cố ý làm hỏng đúng chỗ nó bảo vệ. Một test có thể xanh vì lý do sai, như test 401 vẫn xanh khi mọi request đều bị chặn.

</details>
